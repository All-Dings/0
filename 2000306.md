## Install

### Ubuntu

```bash
$ curl -fsSL https://tailscale.com/install.sh | sh
$ sudo tailscale up --ssh

```

### Mac

Unter "Privacy & Security → Full Disk Access": Add Terminal-Application

```bash
$ sudo systemsetup -setremotelogin on
$ brew install tailscale
$ sudo /opt/homebrew/opt/tailscale/bin/tailscaled
$ tailscale up --ssh
```

### SSH-Tunnel

```bash
$ ssh -N \
  -L 11434:localhost:11434 \
  michaelholzheu@100.64.150.6
```

```bash
$ ssh -N \
  -L 0.0.0.0:11434:localhost:11434 \
  michaelholzheu@100.64.150.6
```

Bridge-I:

```bash
ssh -N \
  -L 172.17.0.1:11434:localhost:11434 \
  michaelholzheu@100.64.150.6
```

Auto-SSH:

```bash
$ sudo apt-get update
$ sudo apt-get install -y autossh
$ autossh -M 0 -N \
  -o "ServerAliveInterval 30" \
  -o "ServerAliveCountMax 3" \
  -L 172.17.0.1:11434:localhost:11434 \
  michaelholzheu@100.64.150.6
```

## Systemd

```
#
# Tunnel-Service that should run on Dings-Host
#
# Allows SSH-Tunnel to Mac-OS Laptop
#
# AWS-Setup:
#   sudo cp ollama-tunnel.service /etc/systemd/system/ollama-tunnel.service
#   sudo systemctl daemon-reload
#   sudo systemctl enable --now tailscaled
#   sudo systemctl enable --now ollama-tunnel
#   sudo tailscale up --ssh=false
#
# Mac-OS-Setup:
#
#   sudo systemsetup -setremotelogin on
#   rew install tailscale
#   sudo /opt/homebrew/opt/tailscale/bin/tailscaled
#   sudo tailscale up
#

[Unit]
Description=Ollama SSH Tunnel (AWS → Mac)
After=network-online.target tailscaled.service
Wants=network-online.target tailscaled.service

[Service]
User=ubuntu
ExecStart=/usr/bin/autossh -M 0 -N \
  -o IdentityFile=/home/ubuntu/.ssh/id_ed25519 \
  -o BatchMode=yes \
  -o StrictHostKeyChecking=accept-new \
  -o ServerAliveInterval=30 \
  -o ServerAliveCountMax=3 \
  -o ExitOnForwardFailure=yes \
  -L 172.17.0.1:11434:localhost:11434 \
  michaelholzheu@100.64.150.6
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
```
